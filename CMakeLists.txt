cmake_minimum_required(VERSION 3.16)
project(vehicle_monitor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(Threads REQUIRED)
add_library(vehicle_core src/core.cpp)
target_include_directories(vehicle_core PUBLIC include)
target_compile_options(vehicle_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(vehicle_core PUBLIC Threads::Threads)
add_executable(vehicle-monitor src/main.cpp)
target_link_libraries(vehicle-monitor PRIVATE vehicle_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(core-tests tests/core_tests.cpp)
 target_link_libraries(core-tests PRIVATE vehicle_core)
 add_test(NAME core COMMAND core-tests)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME runtime COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_tests.py $<TARGET_FILE:vehicle-monitor>)
 endif()
 add_test(NAME simulation COMMAND vehicle-monitor --simulate --duration 1)
endif()
install(TARGETS vehicle-monitor RUNTIME DESTINATION bin)
install(FILES config/vehicle.conf DESTINATION etc/vehicle-monitor)
