cmake_minimum_required(VERSION 3.16)
project(vehicle_monitor VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
file(GLOB_RECURSE VEHICLE_VERSION_INPUTS CONFIGURE_DEPENDS
 "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp"
 "${CMAKE_CURRENT_SOURCE_DIR}/gui/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/gui/*.c"
 "${CMAKE_CURRENT_SOURCE_DIR}/gui/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/config/*"
 "${CMAKE_CURRENT_SOURCE_DIR}/cmake/*")
list(APPEND VEHICLE_VERSION_INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt")
list(SORT VEHICLE_VERSION_INPUTS)
set(VEHICLE_HASH_INPUT "lvgl:0019fc541f759b3323add63034502b0248afc58f\n")
foreach(INPUT IN LISTS VEHICLE_VERSION_INPUTS)
 file(SHA256 "${INPUT}" HASH)
 file(RELATIVE_PATH RELATIVE_INPUT "${CMAKE_CURRENT_SOURCE_DIR}" "${INPUT}")
 string(APPEND VEHICLE_HASH_INPUT "${RELATIVE_INPUT}:${HASH}\n")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${VEHICLE_VERSION_INPUTS})
string(SHA256 VEHICLE_SOURCE_HASH "${VEHICLE_HASH_INPUT}")
string(SUBSTRING "${VEHICLE_SOURCE_HASH}" 0 12 VEHICLE_SOURCE_SHORT)
execute_process(COMMAND git rev-parse --short=12 HEAD WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" OUTPUT_VARIABLE VEHICLE_GIT_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NOT VEHICLE_GIT_REVISION)
 set(VEHICLE_GIT_REVISION "unavailable")
endif()
configure_file(cmake/version.hpp.in generated/vehicle/version.hpp @ONLY)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version.json" "{\"version\":\"${PROJECT_VERSION}+${VEHICLE_SOURCE_SHORT}\",\"source_sha256\":\"${VEHICLE_SOURCE_HASH}\",\"git_base\":\"${VEHICLE_GIT_REVISION}\",\"architecture\":\"${CMAKE_SYSTEM_PROCESSOR}\"}\n")
find_package(Threads REQUIRED)
add_library(vehicle_core src/core.cpp src/acquisition.cpp src/diagnosis.cpp src/storage.cpp src/can_node.cpp src/commands.cpp src/network.cpp src/runtime.cpp)
target_include_directories(vehicle_core PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(vehicle_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(vehicle_core PUBLIC Threads::Threads)
add_executable(vehicle-monitor src/main.cpp)
target_link_libraries(vehicle-monitor PRIVATE vehicle_core)
add_executable(vehicle-input-probe tools/input_probe.cpp)
add_executable(vehicle-log-export tools/log_export.cpp)
target_link_libraries(vehicle-log-export PRIVATE vehicle_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(runtime-unit-tests tests/runtime_tests.cpp)
 target_link_libraries(runtime-unit-tests PRIVATE vehicle_core)
 add_test(NAME runtime-unit COMMAND runtime-unit-tests)
 add_executable(communication-tests tests/communication_tests.cpp)
 target_link_libraries(communication-tests PRIVATE vehicle_core)
 add_test(NAME communication COMMAND communication-tests)
 add_executable(storage-tests tests/storage_tests.cpp)
 target_link_libraries(storage-tests PRIVATE vehicle_core)
 add_test(NAME storage COMMAND storage-tests)
 add_executable(acquisition-tests tests/acquisition_tests.cpp)
 target_link_libraries(acquisition-tests PRIVATE vehicle_core)
 add_test(NAME acquisition COMMAND acquisition-tests)
 add_executable(core-tests tests/core_tests.cpp)
 target_link_libraries(core-tests PRIVATE vehicle_core)
 add_test(NAME core COMMAND core-tests)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING)
  add_test(NAME network-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/network_integration.py $<TARGET_FILE:vehicle-monitor>)
  set_tests_properties(network-integration PROPERTIES TIMEOUT 35)
  add_test(NAME storage-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/storage_integration.py $<TARGET_FILE:vehicle-monitor> $<TARGET_FILE:vehicle-log-export>)
  add_test(NAME runtime COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_tests.py $<TARGET_FILE:vehicle-monitor>)
 endif()
 add_test(NAME simulation COMMAND vehicle-monitor --simulate --duration 1)
endif()
install(TARGETS vehicle-monitor vehicle-log-export vehicle-input-probe RUNTIME DESTINATION bin)
install(FILES config/vehicle.conf DESTINATION etc/vehicle-monitor)

install(PROGRAMS tools/monitor_server.py tools/can_simulator.py DESTINATION share/vehicle-monitor/tools)

option(VEHICLE_BUILD_GUI "Build pinned LVGL 8.2 GUI" ON)
if(VEHICLE_BUILD_GUI)
 file(GLOB_RECURSE LVGL_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lvgl/src/*.c")
 add_library(vehicle_lvgl STATIC ${LVGL_SOURCES} gui/fonts/vehicle_ui_16.c)
 target_include_directories(vehicle_lvgl PUBLIC third_party/lvgl gui)
 target_compile_definitions(vehicle_lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
 target_link_libraries(vehicle_lvgl PUBLIC m)
 add_library(vehicle_gui gui/window.cpp gui/service.cpp)
 target_include_directories(vehicle_gui PUBLIC include)
 target_link_libraries(vehicle_gui PUBLIC vehicle_core vehicle_lvgl)
 set(VEHICLE_GUI_HAS_SDL 0)
 if(NOT CMAKE_CROSSCOMPILING)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
   pkg_check_modules(SDL2 QUIET IMPORTED_TARGET sdl2)
   if(SDL2_FOUND)
    set(VEHICLE_GUI_HAS_SDL 1)
    target_link_libraries(vehicle_gui PRIVATE PkgConfig::SDL2)
    target_compile_definitions(vehicle_gui PRIVATE VEHICLE_GUI_SDL)
   endif()
  endif()
 endif()
 target_link_libraries(vehicle-monitor PRIVATE vehicle_gui)
 target_compile_definitions(vehicle-monitor PRIVATE VEHICLE_BUILD_GUI)
 if(BUILD_TESTING)
  add_executable(gui-tests tests/gui_tests.cpp)
  target_link_libraries(gui-tests PRIVATE vehicle_gui)
  add_test(NAME gui COMMAND gui-tests ${CMAKE_CURRENT_BINARY_DIR}/gui-captures)
  if(Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING)
   add_test(NAME gui-runtime COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/gui_runtime.py $<TARGET_FILE:vehicle-monitor> ${VEHICLE_GUI_HAS_SDL})
  endif()
 endif()
endif()

install(FILES third_party/lvgl/LICENCE.txt gui/fonts/NOTO-COPYRIGHT.txt DESTINATION share/vehicle-monitor/licenses)

if(BUILD_TESTING AND Python3_Interpreter_FOUND AND NOT CMAKE_CROSSCOMPILING)
 if(VEHICLE_BUILD_GUI)
  set(VEHICLE_HEALTH_GUI 1)
 else()
  set(VEHICLE_HEALTH_GUI 0)
 endif()
 add_test(NAME deployment COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/deployment_tests.py ${CMAKE_CURRENT_BINARY_DIR})
 set_tests_properties(deployment PROPERTIES TIMEOUT 90 RUN_SERIAL TRUE)
 add_test(NAME health-runtime COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/health_runtime.py $<TARGET_FILE:vehicle-monitor> ${VEHICLE_HEALTH_GUI})
 set_tests_properties(health-runtime PROPERTIES TIMEOUT 60)
endif()
